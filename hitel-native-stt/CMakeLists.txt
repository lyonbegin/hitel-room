cmake_minimum_required(VERSION 3.16)
project(hitel_native_stt LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_library(hitel_native_stt STATIC src/native_stt.cpp)
target_include_directories(hitel_native_stt PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(hitel_native_stt PRIVATE -Wall -Wextra -Werror -pedantic)
option(HITEL_WITH_SHERPA "Build actual ONNX model inference against public sherpa-onnx C API" OFF)
if(HITEL_WITH_SHERPA)
  if(NOT SHERPA_HEADER_ROOT OR NOT SHERPA_C_API_LIBRARY)
    message(FATAL_ERROR "Supply SHERPA_HEADER_ROOT and SHERPA_C_API_LIBRARY")
  endif()
  add_library(hitel_native_sherpa STATIC src/sherpa_backend.cpp)
  target_include_directories(hitel_native_sherpa PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/include ${SHERPA_HEADER_ROOT})
  target_link_libraries(hitel_native_sherpa PUBLIC hitel_native_stt ${SHERPA_C_API_LIBRARY})
  target_compile_options(hitel_native_sherpa PRIVATE -Wall -Wextra -Werror -pedantic)
  add_executable(hitel_native_stt_cli src/native_stt_cli.cpp)
  target_link_libraries(hitel_native_stt_cli PRIVATE hitel_native_sherpa)
endif()
option(HITEL_BUILD_JNI "Build optional experimental ARM64 Android JNI bridge" OFF)
if(HITEL_BUILD_JNI)
  if(NOT HITEL_WITH_SHERPA)
    message(FATAL_ERROR "HITEL_BUILD_JNI requires public Sherpa C API runtime")
  endif()
  add_library(hitel-native-stt-jni SHARED src/native_stt_jni.cpp)
  target_link_libraries(hitel-native-stt-jni PRIVATE hitel_native_sherpa)
  target_include_directories(hitel-native-stt-jni PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
  if(NOT ANDROID)
    find_package(JNI REQUIRED)
    target_include_directories(hitel-native-stt-jni PRIVATE ${JNI_INCLUDE_DIRS})
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(hitel_native_stt_tests tests/native_stt_tests.cpp)
  target_link_libraries(hitel_native_stt_tests PRIVATE hitel_native_stt)
  target_compile_options(hitel_native_stt_tests PRIVATE -Wall -Wextra -Werror -pedantic)
  add_test(NAME hitel_native_stt_engine COMMAND hitel_native_stt_tests)
endif()
